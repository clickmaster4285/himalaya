// ==================== MAIN PARSER (Enhanced) ====================
const parsePasContent = (pasText: string): PasContent => {
  const lines = pasText.split('\n').filter(Boolean);
  let before = '';
  let after = '';
  let problem = '';
  let agitation = '';
  let solution = '';
  let currentSection = 'before';
  const clickItems: ClickItem[] = [];

  const icons: Record<string, any> = {
    'C': Target, 'L': Shield, 'I': Sparkles, 'C2': GitBranch, 'K': Users, 'S': Rocket,
  };

  // Check for new KEY ANCHORS style
  const hasKeyAnchors = lines.some(l => l.trim().startsWith('KEY ANCHORS:'));

  if (hasKeyAnchors) {
    const fullText = lines.join('\n');
    const sections: Record<string, string> = {};

    // Split by major headers
    const headerMatches = [...fullText.matchAll(/(KEY ANCHORS:|COMPLIANCE:|CPCs:|Solution:)/gi)];
    
    let lastIndex = 0;
    for (let i = 0; i < headerMatches.length; i++) {
      const match = headerMatches[i];
      const header = match[0].trim();
      const start = match.index!;
      const end = headerMatches[i + 1] ? headerMatches[i + 1].index! : fullText.length;

      const content = fullText.slice(start + header.length, end).trim();
      sections[header.replace(':', '').trim()] = content;

      lastIndex = end;
    }

    // Map to PAS
    if (sections['KEY ANCHORS']) {
      problem = `Major anchors: ${sections['KEY ANCHORS']}`;
    }
    if (sections['COMPLIANCE'] || sections['CPCs']) {
      agitation = [
        sections['COMPLIANCE'],
        sections['CPCs']
      ].filter(Boolean).join('\n\n');
    }
    if (sections['Solution']) {
      solution = sections['Solution'];
    }

    return { before, after, problem, agitation, solution, clickItems };
  }

  // === Fallback: Original parsing logic (unchanged behavior) ===
  for (const line of lines) {
    const trimmed = line.trim();

    if (trimmed.startsWith('PROBLEM:')) {
      problem = trimmed.replace('PROBLEM:', '').trim();
      currentSection = 'problem';
      continue;
    }
    if (trimmed.startsWith('AGITATION:')) {
      agitation = trimmed.replace('AGITATION:', '').trim();
      currentSection = 'agitation';
      continue;
    }
    if (trimmed.startsWith('SOLUTION:')) {
      solution = trimmed.replace('SOLUTION:', '').trim();
      currentSection = 'solution';
      continue;
    }
    if (trimmed.startsWith('THE BEFORE:')) {
      before = trimmed.replace('THE BEFORE:', '').trim();
      currentSection = 'before';
      continue;
    }
    if (trimmed.startsWith('THE AFTER:')) {
      after = trimmed.replace('THE AFTER:', '').trim();
      currentSection = 'after';
      continue;
    }

    if (trimmed.includes('CLICK GROWTH SYSTEM') || trimmed.includes('THE BRIDGE')) {
      currentSection = 'bridge';
      continue;
    }

    if (currentSection === 'bridge' && trimmed.match(/^[CKLIS] —/)) {
      const match = trimmed.match(/^([CKLIS]) — ([^:]+):/);
      if (match) {
        const letter = match[1];
        const title = match[2].trim();
        const desc = trimmed.replace(match[0], '').trim();
        const iconKey = letter === 'C' && clickItems.some(item => item.letter === 'C') ? 'C2' : letter;

        clickItems.push({ letter, title, desc, icon: icons[iconKey] || Target });
      }
      continue;
    }

    // Multi-line handling
    if (currentSection === 'bridge' && trimmed && clickItems.length > 0 &&
        !trimmed.match(/^(THE|PROBLEM|AGITATION|SOLUTION|KEY ANCHORS|COMPLIANCE|CPCs)/i)) {
      clickItems[clickItems.length - 1].desc += ' ' + trimmed;
    }

    if (currentSection === 'problem') problem += ' ' + trimmed;
    if (currentSection === 'agitation') agitation += ' ' + trimmed;
    if (currentSection === 'solution') solution += ' ' + trimmed;
  }

  return {
    before: before.trim(),
    after: after.trim(),
    problem: problem.trim(),
    agitation: agitation.trim(),
    solution: solution.trim(),
    clickItems,
  };
};

// Main Export

      <PasLightSection pasText={pasText} />
    </>
